## Архитектурные решения

### D1 — ShopScreen класс

`core/progression/shop_screen.gd`:
```gdscript
class_name ShopScreen extends RefCounted
var _offered_item_ids: Array[StringName] = []  # 3 ids
var _discount: float = 0.5  # MAP_MERCHANT_DISCOUNT

func refresh(_unlocked_units: Array = []) -> void:
    var item_ids: Array = ContentDB_static.get_all_ids_for_type("items")
    if item_ids.is_empty():
        return
    _offered_item_ids.clear()
    for i in 3:
        if item_ids.is_empty():
            break
        var idx: int = int(Rng.randf_range(0.0, float(item_ids.size())))
        if idx >= item_ids.size():
            idx = item_ids.size() - 1
        _offered_item_ids.append(item_ids[idx])

func get_offered_count() -> int: return _offered_item_ids.size()
func get_item_id(slot: int) -> StringName: return _offered_item_ids[slot] if 0 <= slot < _offered_item_ids.size() else &""
func get_discounted_price(slot: int) -> int:
    var def = ContentDB_static.get_by_id(_offered_item_ids[slot])
    return int(round(float(def.cost) * _discount)) if def != null else 0
```

### D2 — RunController.buy_item API

```gdscript
## Покупает item из shop slot (с MAP_MERCHANT_DISCOUNT). Returns true / false.
func buy_item(slot: int) -> bool:
    if slot < 0 or slot >= shop.get_offered_count():
        return false
    var id: StringName = shop.get_item_id(slot)
    var price: int = shop.get_discounted_price(slot)
    if state.gold < price:
        return false
    if not grant_item(id):
        return false
    state.gold -= price
    GameBus.emit_gold_changed(state.gold)
    GameLog.info("run", "Shop buy", {"id": id, "price": price, "gold_left": state.gold})
    return true
```
